Kirby::plugin(
	name: 'my/markdown',
	extends: [
		'components' => [
			'markdown' => function (Kirby $kirby, string $text = null, array $options = [], bool $inline = false) {
				return YourMarkdownParser::parse($text);
			}
		]
	]
);
